ALTER TABLE fluxo_versionado
    ADD COLUMN IF NOT EXISTS atual BOOLEAN NOT NULL DEFAULT FALSE;

WITH versoes_rankeadas AS (
    SELECT
        id,
        ROW_NUMBER() OVER (
            PARTITION BY fluxo_id
            ORDER BY versao DESC, id DESC
        ) AS linha
    FROM fluxo_versionado
)
UPDATE fluxo_versionado fv
SET atual = vr.linha = 1
FROM versoes_rankeadas vr
WHERE fv.id = vr.id;
